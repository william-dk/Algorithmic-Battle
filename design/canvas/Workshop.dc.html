1import { Agent, Observation, Action } from "@arena/sdk";
2
3// 直取大本营：选「花费 + 0.3 × 到敌方大本营的距离」最小的格子，
4// 买不起就先攒能量。
5export default class Rusher implements Agent {
6act(obs: Observation): Action[] {
7let best = -1;
8let bestScore = Infinity;
9
10for (const cell of obs.frontier) {
11const score = obs.costOf(cell) + 0.3 * obs.distToEnemyBase(cell);
12if (score < bestScore) {
13bestScore = score;
14best = cell;
15}
16}
17
18if (best < 0 || obs.costOf(best) > obs.energy) return [];
19const type = obs.owner[best] === 0 ? "claim" : "attack";
20return [{ type, cell: best }];
21}
22}
